name: "paired-bridge"
nodes: {
    name: "host1"
    vendor: HOST
    config: {
        image: "alpine:latest"
        command: ["/bin/sh", "-c", "ip addr add 192.168.1.1/24 dev eth1 && ip link set eth1 up && sleep infinity"]
    }
}
nodes: {
    name: "bridge-server"
    vendor: FORWARD
    config: {
        image: "us-west1-docker.pkg.dev/kne-external/kne/bridge:latest"
    }
    # Expose the packet bridge service externally via a K8s LoadBalancer Service.
    # Required when connecting a bridge client from outside the cluster (e.g. Borg or workstation).
    services: {
        key: 50058
        value: {
            name: "wire"
            inside: 50058
        }
    }
}
nodes: {
    name: "bridge-client"
    vendor: FORWARD
    config: {
        image: "us-west1-docker.pkg.dev/kne-external/kne/bridge:latest"
        args: ["client", "--peer=service-bridge-server:50058", "--interface=eth1"]
    }
}
nodes: {
    name: "host2"
    vendor: HOST
    config: {
        image: "alpine:latest"
        command: ["/bin/sh", "-c", "ip addr add 192.168.1.2/24 dev eth1 && ip link set eth1 up && sleep infinity"]
    }
}

links: {
    a_node: "host1"
    a_int: "eth1"
    z_node: "bridge-server"
    z_int: "eth1"
}
links: {
    a_node: "bridge-client"
    a_int: "eth1"
    z_node: "host2"
    z_int: "eth1"
}
